def digest(data):

def normalized_relative(value):
    if not isinstance(value, str) or not value or "\x00" in value:
        raise GateError("MALFORMED_EDIT_SPEC")
    value = unicodedata.normalize("NFC", value.replace("\\", "/"))
    parts = value.split("/")
    if value.startswith("/") or any(part in ("", ".", "..") for part in parts):

def open_absolute_dir(path):
    if not os.path.isabs(path):
        raise GateError("WORKSPACE_NOT_ABSOLUTE")
    current = os.open("/", os.O_RDONLY | os.O_DIRECTORY)
    try:
        for part in [p for p in path.split("/") if p]:
            nxt = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=current)
            os.close(current)

            meta = entry.stat(follow_symlinks=False)

                protected.add(rel)

                child = os.open(entry.name, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=dir_fd)
                try:
                    collect_tree(child, rel, protected, regulars)
                finally:

            elif regulars and stat.S_ISREG(meta.st_mode):

def collect_path(root_fd, rel, protected):
    rel = normalized_relative(rel)
    parts = rel.split("/")
    parent = open_relative_dir(root_fd, "/".join(parts[:-1]), missing_ok=True)
    if parent is None:
        protected.add(rel)
        return
    try:
        try:
            meta = os.stat(parts[-1], dir_fd=parent, follow_symlinks=False)
        except FileNotFoundError:
            protected.add(rel)
            return
        if stat.S_ISDIR(meta.st_mode):
            child = os.open(parts[-1], os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=parent)
            try:
                collect_tree(child, rel, protected, True)
            finally:
                os.close(child)
        else:
            protected.add(rel)
    finally:
        os.close(parent)


def status_paths(workspace):
    raw = run_git(
        workspace,
        "status",
        "--porcelain=v1",
        "-z",
        "--untracked-files=all",
        "--ignored=matching",
    ).stdout
    fields = raw.split(b"\x00")
    protected = []
    index = 0
    while index < len(fields) and fields[index]:
        field = fields[index]
        if len(field) < 4 or field[2:3] != b" ":
            raise GateError("UNPARSABLE_GIT_STATUS")
        xy = field[:2].decode("ascii", "strict")
        path = field[3:].decode("utf-8", "surrogateescape")
        protected.append(path[:-1] if path.endswith("/") else path)
        if "R" in xy or "C" in xy:
            index += 1
            if index >= len(fields) or not fields[index]:
                raise GateError("UNPARSABLE_GIT_STATUS")
            path = fields[index].decode("utf-8", "surrogateescape")
            protected.append(path[:-1] if path.endswith("/") else path)
        index += 1
    return protected


def leaf_record(root_fd, rel):
    parts = rel.split("/")
    parent = open_relative_dir(root_fd, "/".join(parts[:-1]), missing_ok=True)
    if parent is None:
        return {"path": rel, "type": "missing"}
    try:
        try:
            meta = os.stat(parts[-1], dir_fd=parent, follow_symlinks=False)
        except FileNotFoundError:
            return {"path": rel, "type": "missing"}
        base = {"path": rel, "mode": stat.S_IMODE(meta.st_mode), "uid": meta.st_uid, "size": meta.st_size}
        if stat.S_ISLNK(meta.st_mode):
            base.update(type="symlink", target=os.readlink(parts[-1], dir_fd=parent))
            return base
        if not stat.S_ISREG(meta.st_mode):
            base.update(type="other")
            return base
        fd = os.open(parts[-1], os.O_RDONLY | os.O_NOFOLLOW, dir_fd=parent)
        try:
            h = hashlib.sha256()
            while True:
                chunk = os.read(fd, 1024 * 1024)
                if not chunk:
                    break
                h.update(chunk)
            base.update(type="regular", sha256=h.hexdigest())
            return base
        finally:
            os.close(fd)
    finally:
        os.close(parent)


def is_prefix(left, right):
    left_parts = left.split("/")
    right_parts = right.split("/")
    return len(left_parts) <= len(right_parts) and left_parts == right_parts[: len(left_parts)]


def verify_specs(root_fd, specs, protected):
    if not isinstance(specs, list):
        raise GateError("MALFORMED_EDIT_SPECS")
    normalized = []
    aliases = set()
    for item in specs:
        if not isinstance(item, dict) or set(item) != {"kind", "path"}:
            raise GateError("MALFORMED_EDIT_SPEC")
        kind = item["kind"]
        if kind not in ("literal", "create_only_prefix"):
            raise GateError("MALFORMED_EDIT_SPEC")
        path = normalized_relative(item["path"])
        alias = unicodedata.normalize("NFC", path).casefold()
        if alias in aliases:
            raise GateError("DUPLICATE_OR_ALIAS_EDIT_SPEC", path)
        aliases.add(alias)
        parent = "/".join(path.split("/")[:-1])
        try:
            fd = open_relative_dir(root_fd, parent, missing_ok=True)
        except OSError:
            raise GateError("SYMLINK_OR_NON_DIRECTORY_ANCESTOR", path)
        if fd is not None:
            os.close(fd)
        normalized.append({"kind": kind, "path": path})
    if normalized != sorted(normalized, key=lambda value: (value["path"], value["kind"])):
        raise GateError("NONCANONICAL_EDIT_SPEC_ORDER")
    for item in normalized:
        for leaf in protected:
            if is_prefix(item["path"], leaf) or is_prefix(leaf, item["path"]):
                raise GateError("PROTECTED_ALLOWLIST_OVERLAP", leaf)
    return normalized


def secure_child(parent_fd, name, create):
    if create:
        try:
            os.mkdir(name, 0o700, dir_fd=parent_fd)
        except FileExistsError:
            pass
    fd = os.open(name, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=parent_fd)

        os.close(fd)
        raise GateError("UNSAFE_EVIDENCE_DIRECTORY", name)
    return fd

def execute(args):

    try:
        receipt_bytes = read_relative(workspace_fd, args.approval_receipt_path)
        if digest(receipt_bytes) != args.expected_receipt_sha256:

        try:

        except (UnicodeDecodeError, json.JSONDecodeError):
            raise GateError("INVALID_APPROVAL_RECEIPT", args.approval_receipt_path)
        if receipt.get("schema") != SCHEMA or receipt.get("plan_path") != args.plan_path:

        if digest(plan_bytes) != receipt.get("plan_sha256") or len(plan_bytes) != receipt.get("plan_bytes"):

            raise GateError("G0_SOURCE_DIGEST_MISMATCH")
        reviews = receipt.get("planning_reviews")
        required = {("momus", "OKAY", True), ("independent", "OKAY", True)}

        }

            raise GateError("PLANNING_APPROVAL_MISSING")

        if run_git(args.workspace_root, "diff", "--cached", "--quiet", "--ignore-submodules", "--", check=False).returncode != 0:
            raise GateError("PREEXISTING_STAGED_CONTENT")

        os.close(workspace_fd)

## Commit strategy
- Use one atomic commit per Todo using the declared message, but only the primary coordinator may touch the index or commit. Parallel workers stop after disjoint edits/tests/evidence. At the wave barrier the coordinator invokes the Todo-3 lock helper, which atomically acquires `.git/psi-v2-index.lock`, aborts on pre-existing staged files or allowlist collision, verifies Todo-1 protected hashes, stages one literal allowlist, shows cached diff, commits, clears the index, and only then processes the next Todo.
- A Todo is eligible for coordinator commit only after both literal happy/failure commands, targeted/strict gates, evidence, and post-edit status pass. Never use `git add .`, `git add -A`, broad globs, cleanup, destructive reset/checkout, per-worker alternate index commits, or concurrent index access.
- Existing modified/untracked files belong to the user. Never include `.env*` secrets, raw business workbooks, `input/`, `old_check/`, `outputs/`, `.tmp/`, `.codex-tmp/`, generated Parquet/XLSX, node_modules, caches, or unrelated changes in a commit.
- If a task touches a path changed by another worker, pause only that task, re-read the file/status, and adapt without reverting. Parallel tasks own disjoint allowlists wherever the dependency matrix permits.
- Preferred execution is the current protected checkout because critical reference material is untracked; a dedicated worktree is safe only after Todo 1 has materialized the sanitized, hash-locked baseline that the worktree needs. No branch, PR, push, deployment, or production mutation is implied by this plan.
- Todo 42/43 implementation and docs are committed before final evidence is stamped. The no-edit finalization barrier then compares the live plan to the independently reviewed digest in the approval receipt, verifies protected paths, runs the unchanged release gate, runs the unchanged read-only production preflight, seals one prereview index containing both, runs and verifies F1-F4, then seals a separate approval index containing their four report hashes. Any later source/plan/receipt/protected-tree/index/commit drift invalidates both indexes and all approvals and restarts the barrier; a new live digest is never accepted without fresh dual plan review.
